cmake_minimum_required(VERSION 3.15)
project(AddonHttps LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_compile_definitions(_WIN32_WINNT=0x0501 WINVER=0x0501 _USING_V110_SDK71_ _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
set(MBEDTLS_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/config.h" CACHE STRING "" FORCE)
add_subdirectory("${DEPENDENCIES}/mbedtls-3.6.7" mbedtls)
add_library(AddonHttps STATIC ../https_client.cpp)
target_include_directories(AddonHttps PRIVATE "${GENERATED}")
target_link_libraries(AddonHttps PRIVATE mbedtls mbedx509 mbedcrypto ws2_32 advapi32)
target_compile_options(AddonHttps PRIVATE /W4 /EHsc /utf-8 /Zc:threadSafeInit-)
